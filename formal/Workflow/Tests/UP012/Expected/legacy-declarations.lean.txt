theorem dropSigner_claim :
∀ p,
      validScriptContext (claim p) = true →
      accepts ledgerFuel (claim p) →
      (signatoriesOf (claim p)) ≠ [] →
      validScriptContext (withSignatories (claim p) (signatoriesOf (claim p)).tail) = true →
      rejects ledgerFuel (withSignatories (claim p) (signatoriesOf (claim p)).tail) := by blaster

theorem dropSigner_claim_applies :
∃ p,
      validScriptContext (claim p) = true ∧
      accepts ledgerFuel (claim p) ∧
      (signatoriesOf (claim p)) ≠ [] ∧
      validScriptContext (withSignatories (claim p) (signatoriesOf (claim p)).tail) = true := by blaster
